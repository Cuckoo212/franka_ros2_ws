cmake_minimum_required(VERSION 3.16)
project(cable_moveit_grasp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(DEPS rclcpp rclcpp_action geometry_msgs sensor_msgs std_msgs std_srvs
  visualization_msgs shape_msgs interactive_markers tf2 tf2_ros tf2_geometry_msgs
  moveit_ros_planning_interface moveit_core moveit_msgs franka_msgs)
foreach(dep IN LISTS DEPS)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(cable_moveit_grasp_node src/cable_moveit_grasp_node.cpp)
target_include_directories(cable_moveit_grasp_node PRIVATE include)
target_link_libraries(cable_moveit_grasp_node Eigen3::Eigen)
ament_target_dependencies(cable_moveit_grasp_node ${DEPS})
install(TARGETS cable_moveit_grasp_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cloud_calibration_node.py DESTINATION lib/${PROJECT_NAME} RENAME cloud_calibration_node)
install(FILES scripts/calibration_math.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME cloud_callback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_cloud_callback.py)
  add_test(NAME calibration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_calibration.py)
  add_executable(test_geometry test/test_geometry.cpp)
  target_include_directories(test_geometry PRIVATE include)
  target_link_libraries(test_geometry Eigen3::Eigen)
  add_test(NAME geometry COMMAND test_geometry)
  add_executable(test_path_validation test/test_path_validation.cpp)
  target_include_directories(test_path_validation PRIVATE include)
  target_link_libraries(test_path_validation Eigen3::Eigen)
  ament_target_dependencies(test_path_validation ${DEPS})
  add_test(NAME path_validation COMMAND test_path_validation)
  add_executable(test_platform test/test_platform.cpp)
  target_include_directories(test_platform PRIVATE include)
  target_link_libraries(test_platform Eigen3::Eigen)
  ament_target_dependencies(test_platform geometry_msgs shape_msgs)
  add_test(NAME platform COMMAND test_platform)
endif()
ament_package()
